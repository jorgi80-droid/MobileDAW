# JNI library of the Android app: the shared C++ engine (native/) + Oboe output (bridge.cpp)
cmake_minimum_required(VERSION 3.18)
project(mobiledaw_android CXX)
set(CMAKE_CXX_STANDARD 17)
add_subdirectory(${CMAKE_CURRENT_SOURCE_DIR}/../../../../../native ${CMAKE_CURRENT_BINARY_DIR}/engine)
find_package(oboe REQUIRED CONFIG)
add_library(mobiledaw SHARED bridge.cpp)
target_compile_options(mobiledaw PRIVATE -O2 -ffp-contract=off)
target_link_libraries(mobiledaw mdaw_engine oboe::oboe log)
